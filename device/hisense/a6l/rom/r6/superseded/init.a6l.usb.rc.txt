# A6L USB device mode for the installed ROM (r6; completeness audit 29 Sep 2026). REPLACES rom/init/init.a6l.usb.rc.
# Why: the r1-r5 ROM never creates the configfs gadget (/config/usb_gadget/g1): AOSP init.usb.configfs.rc only links
# functions into an existing gadget, and its MTP/PTP rules link "mtp.gs0", an Android-kernel-only function that the
# mainline kernel does not have. Result on an installed r5: no ADB and no MTP over USB (never seen because every ROM test
# ran in QEMU or inside the recovery). This file builds the gadget like the Pixel/Linaro configfs devices:
#   ffs.adb  (adbd), ffs.mtp / ffs.ptp (FunctionFS; system_server's UsbDeviceManager writes the MTP descriptors to
#   /dev/usb-ffs/mtp/ep0 at start, MtpServer serves it), rndis.gs4 / ncm.gs5 (USB tethering), midi.gs6.
# sys.usb.configfs=2 = "vendor owns configfs": system init.usb.configfs.rc (configfs=1 rules, mtp.gs0) stays idle and
# every function set the framework can request is handled below. The framework's legacy USB handler (no gadget HAL in
# this product) drives sys.usb.config; connection state comes from /sys/class/udc/<udc>/state (r6 overlay
# config_enableUdcSysfsUsbStateUpdate + genfs sysfs_udc), because mainline has no android_usb uevents.
# Kernel (r5 config): USB_CONFIGFS_F_FS/RNDIS/NCM/F_MIDI=y. The kernel runs with a6l_manual_usb=1: D+ pull-up is released
# with soft_connect after every (re)bind (as rom/init/init.a6l.usb.rc did).
# RAM-container guard (realinit LineageOS inside the recovery): nothing here touches configfs when ro.boot.a6l.container=1.

on early-init
    setprop vendor.a6l.usb.managed 1

on early-init && property:ro.boot.a6l.container=1
    setprop vendor.a6l.usb.managed 0

on early-boot && property:vendor.a6l.usb.managed=1
    mkdir /config/usb_gadget/g1 0770 shell shell
    write /config/usb_gadget/g1/idVendor 0x18D1
    write /config/usb_gadget/g1/idProduct 0x4EE7
    write /config/usb_gadget/g1/bcdDevice 0x0223
    write /config/usb_gadget/g1/bcdUSB 0x0200
    write /config/usb_gadget/g1/os_desc/use 1
    write /config/usb_gadget/g1/os_desc/b_vendor_code 0x1
    write /config/usb_gadget/g1/os_desc/qw_sign "MSFT100"
    mkdir /config/usb_gadget/g1/strings/0x409 0770
    write /config/usb_gadget/g1/strings/0x409/serialnumber ${ro.serialno}
    write /config/usb_gadget/g1/strings/0x409/manufacturer ${ro.product.manufacturer}
    write /config/usb_gadget/g1/strings/0x409/product ${ro.product.model}
    mkdir /config/usb_gadget/g1/configs/b.1 0770 shell shell
    mkdir /config/usb_gadget/g1/configs/b.1/strings/0x409 0770 shell shell
    write /config/usb_gadget/g1/configs/b.1/MaxPower 500
    symlink /config/usb_gadget/g1/configs/b.1 /config/usb_gadget/g1/os_desc/b.1
    mkdir /config/usb_gadget/g1/functions/ffs.adb
    mkdir /config/usb_gadget/g1/functions/ffs.mtp
    mkdir /config/usb_gadget/g1/functions/ffs.ptp
    mkdir /config/usb_gadget/g1/functions/rndis.gs4
    mkdir /config/usb_gadget/g1/functions/ncm.gs5
    mkdir /config/usb_gadget/g1/functions/midi.gs6
    mkdir /dev/usb-ffs 0775 shell system
    mkdir /dev/usb-ffs/adb 0770 shell shell
    mount functionfs adb /dev/usb-ffs/adb rmode=0770,fmode=0660,uid=2000,gid=2000
    mkdir /dev/usb-ffs/mtp 0770 mtp mtp
    mkdir /dev/usb-ffs/ptp 0770 mtp mtp
    mount functionfs mtp /dev/usb-ffs/mtp rmode=0770,fmode=0660,uid=1024,gid=1024,no_disconnect=1
    mount functionfs ptp /dev/usb-ffs/ptp rmode=0770,fmode=0660,uid=1024,gid=1024,no_disconnect=1
    setprop sys.usb.mtp.device_type 3
    setprop sys.usb.controller a800000.usb
    setprop sys.usb.configfs 2

on boot && property:vendor.a6l.usb.managed=1
    setprop sys.usb.controller a800000.usb
    setprop sys.usb.configfs 2

# ---- none: unbind and clear every function link
on property:sys.usb.config=none && property:sys.usb.configfs=2
    write /config/usb_gadget/g1/UDC "none"
    stop adbd
    setprop sys.usb.ffs.ready 0
    rm /config/usb_gadget/g1/configs/b.1/f1
    rm /config/usb_gadget/g1/configs/b.1/f2
    rm /config/usb_gadget/g1/configs/b.1/f3
    setprop sys.usb.state ${sys.usb.config}
