# r5 bug hunt boot-init (29 Sep 2026): kernel features Android userspace needs that the V67 Image builds as MODULES
# (phone-kernel-v67-candidate-20260919/config: CONFIG_FUSE_FS=m, CONFIG_UHID=m) and that no other group loaded.
# Loaded first by a6l-modules.sh display (early-init), long before vold/system_server/the Bluetooth stack start.
#  - fuse: vold MountUserFuse opens /dev/fuse for /storage/emulated (MediaProvider FUSE daemon) and for every public
#    volume (microSD); without it no user storage is mounted.
#  - uhid: /dev/uhid (ueventd 0660 uhid uhid) = the Bluetooth stack's HID host (BT keyboards/mice/gamepads);
#    without the module the node never exists.
fuse.ko
uhid.ko

#  - uid_sys_stats: /proc/uid_cputime, /proc/uid_io, /proc/uid_procstat (BatteryStats per-UID user/sys time, storaged per-UID io).
xt_quota2.ko
uid_sys_stats.ko
